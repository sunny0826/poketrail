#include "adventure_model.hpp"
#include "save_codec.hpp"
#include <cstdio>
#include <vector>

int main(int argc, char** argv) {
  using namespace pokedex::adventure;
  if (argc != 2) return 1;
  Save s; s.sequence = 7; s.bootGeneration = 1;
  s.starter = s.partner = 1; s.rosterRevision = 2;
  setOwned(s, 1); setRegistered(s, 1); setOwned(s, 16); setRegistered(s, 16);
  s.totalSteps = s.undatedSteps = 1000; s.progress[0].walked = 1000;
  s.explorationSpent = 600; s.explorationCredit = 400; s.encounterTarget = 300;
  s.captureCounts[15] = 1;
  Capture& c = s.firstCapture[15]; c.species = 16; c.route = Route::Forest;
  c.at.bootGeneration = 1; c.at.totalSteps = 300;
  s.recent[0] = c; s.recentSize = 1;
  s.route = Route::Coast;
  s.nextEncounterId = 3; s.phase = Phase::ThrowReady; s.result = Result::Won;
  s.encounter.id = 2; s.encounter.species = 72; s.encounter.partner = 1;
  s.encounter.route = Route::Coast; s.encounter.seed = 0x2468ace0u;
  if (s.schema != 6 || !validate(s)) return 2;
  std::vector<uint8_t> bytes(codec::maxEncodedSize); size_t length = 0;
  if (!codec::encode(s, bytes.data(), bytes.size(), length)) return 3;
  FILE* f = std::fopen(argv[1], "wb");
  if (!f) return 4;
  const bool ok = std::fwrite(bytes.data(), 1, length, f) == length;
  return std::fclose(f) == 0 && ok ? 0 : 5;
}
